Pacman

CommandDescription
pacman -Qmlist foreign/AUR packages
pacman -Qnlist official repo packages
sudo pacman -Rns xxxremove a package and its unneeded dependencies
sudo paccache -rdelete old cached package versions, keep the latest
sudo pacman -Sccdelete all cached packages
du -sh /var/cache/pacman/pkg/check the size of the package cache
sudo pacman -Ss <name>search the repos
sudo pacman -Si <name>show info for a repo package
pacman -Qo /path/to/filefind which package owns a file
pacman -Ql <name> | grep '\.so'list the .so files a package provides

CommandDescription

Text & Search

CommandDescription
find /path -name "*.txt"find by name
find /path -iname "*.txt"find by name, case-insensitive
find /path -type f -name "*.txt"files only
find /path -type d -name logsdirectories only
find /path -mtime -7modified in the last 7 days
find /path -size +100Mlarger than 100 MB
find /path -user <name>owned by a given user
find / -perm -4000 2>/dev/nullfind SUID binaries
find / -perm /6000 2>/dev/nullfind SUID or SGID binaries
fd pattern /pathfind files/dirs matching a pattern (fd)
fd -e txtby extension
fd -t f pattern ; fd -t d patternfiles only / directories only
fd -H patterninclude hidden files
fd -I patternignore .gitignore rules
fd -x <command>run a command on each match
grep -rl "text" / --exclude-dir={proc,sys,dev} 2>/dev/nullsearch file contents recursively, skipping virtual filesystems
grep -rlI "text" / 2>/dev/nullsame, skipping binary files
grep -oP "key=\K\S+"extract only the value after a known key (\K drops the match before it)
grep -w 80/tcp /etc/serviceswhole-word match, no partial hits
grep -E '\b(53|332)\b' filematch either of two exact numbers
find / -type f -name xxx 2>/dev/null | xargs grep -l "text" 2>/dev/nullpipe find results into grep via xargs
find / -type f -name xxx -exec grep -l "text" {} \; 2>/dev/nullsame, using -exec instead of xargs
cat $(ls | grep root.txt)cat a file whose name has odd characters, via a glob-free workaround
cat -- "$(ls)"cat a file whose name starts with a dash
